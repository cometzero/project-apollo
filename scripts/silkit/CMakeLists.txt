cmake_minimum_required(VERSION 3.16)
project(apollo_vmcu_silkit LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 14)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(SilKit 5.0.7 EXACT REQUIRED CONFIG)
find_package(Threads REQUIRED)
add_executable(vmcu-silkit participant.cpp)
target_link_libraries(vmcu-silkit PRIVATE SilKit::SilKit Threads::Threads)
target_compile_options(vmcu-silkit PRIVATE -Wall -Wextra -Werror -Wpedantic)
add_executable(can-wire-test can-wire-test.cpp)
target_compile_options(can-wire-test PRIVATE -Wall -Wextra -Werror -Wpedantic)
enable_testing()
add_test(NAME can-wire COMMAND can-wire-test)
