# ​J1.2.742 AArch64_S1TTBR

##### J1.2.742 AArch64\_S1TTBR

```
// AArch64_S1TTBR()
// ================
// Identify stage 1 table base register for the acting translation regime
func AArch64_S1TTBR(regime : Regime, va : bits(64)) => bits(128)
begin
    let varange : VARange = AArch64_GetVARange(va);
    case regime of
        when Regime_EL3 =>  return ZeroExtend{128}(TTBR0_EL3());
        when Regime_EL2 =>  return ZeroExtend{128}(TTBR0_EL2());
        when Regime_EL20 =>
            if varange == VARange_LOWER then
                return ZeroExtend{128}(TTBR0_EL2());
            else
                return ZeroExtend{128}(TTBR1_EL2());
            end;
        when Regime_EL10 =>
            if varange == VARange_LOWER then
                return ZeroExtend{128}(TTBR0_EL1());
            else
                return ZeroExtend{128}(TTBR1_EL1());
            end;
    end;
end;
```
