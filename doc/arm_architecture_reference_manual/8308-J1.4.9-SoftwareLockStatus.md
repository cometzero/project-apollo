##### J1.4.9 SoftwareLockStatus

```
// SoftwareLockStatus()
// ====================
// Returns the state of the Software Lock.
func SoftwareLockStatus(addrdesc : AddressDescriptor) => boolean
begin
    let component : Component = GetAccessComponent(addrdesc);
    if !HaveSoftwareLock(component) then
        return FALSE;
    end;
    case component of
        when Component_ETE   => return TRCLSR().SLK == ‘1’;
        when Component_Debug => return EDLSR().SLK == ‘1’;
        when Component_PMU   => return PMLSR().SLK == ‘1’;
        when Component_CTI   => return CTILSR().SLK == ‘1’;
        otherwise            => return FALSE;
    end;
end;
```
