##### J1.2.268 GetCurrentGCSPointer

```
// GetCurrentGCSPointer()
// ======================
// Returns the value of the current GCS pointer register.
func GetCurrentGCSPointer() => bits(64)
begin
    var ptr : bits(64);
    case PSTATE.EL of
        when EL0 =>
            ptr = GCSPR_EL0().PTR::‘000’;
        when EL1 =>
            ptr = GCSPR_EL1().PTR::‘000’;
        when EL2 =>
            ptr = GCSPR_EL2().PTR::‘000’;
        when EL3 =>
            ptr = GCSPR_EL3().PTR::‘000’;
    end;
    return ptr;
end;
```
