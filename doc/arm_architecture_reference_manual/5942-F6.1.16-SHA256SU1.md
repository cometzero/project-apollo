##### F6.1.16 SHA256SU1

SHA256 schedule update 1.

It has encodings from the following instruction sets: A32 ([A1](/documentation/ddi0487/mc/-Part-F-The-AArch32-Instruction-Sets/-Chapter-F6-T32-and-A32-Advanced-SIMD-and-Floating-point-Instruction-Descriptions/-F6-1-Alphabetical-list-of-Advanced-SIMD-and-floating-point-instructions/-F6-1-16-SHA256SU1?lang=en#isa_sha256su1_iclass_a1)) and T32 ([T1](/documentation/ddi0487/mc/-Part-F-The-AArch32-Instruction-Sets/-Chapter-F6-T32-and-A32-Advanced-SIMD-and-Floating-point-Instruction-Descriptions/-F6-1-Alphabetical-list-of-Advanced-SIMD-and-floating-point-instructions/-F6-1-16-SHA256SU1?lang=en#isa_sha256su1_iclass_t1)).

###### Assembler Symbols

<Qd>
:   ​

    Is the 128-bit name of the SIMD&FP destination register, encoded in the “D:Vd” field as <Qd>\*2.

<Qn>
:   ​

    Is the 128-bit name of the first SIMD&FP source register, encoded in the “N:Vn” field as <Qn>\*2.

<Qm>
:   ​

    Is the 128-bit name of the second SIMD&FP source register, encoded in the “M:Vm” field as <Qm>\*2.

###### Operation

```
if ConditionPassed() then
    EncodingSpecificOperations();
    CheckCryptoEnabled32();
    var elt : bits(32);
    var result : bits(128);
    let x : bits(128) = Q(d>>1);
    let y : bits(128) = Q(n>>1);
    let z : bits(128) = Q(m>>1);
    let T0 : bits(128) = z[31:0] :: y[127:32];
    var T1 : bits(64) = z[127:64];
    for e = 0 to 1 do
        elt = T1[e*:32];
        elt = ROR(elt, 17) XOR ROR(elt, 19) XOR LSR(elt, 10);
        elt = elt + x[e*:32] + T0[e*:32];
        result[e*:32] = elt;
    end;
    T1 = result[63:0];
    for e = 2 to 3 do
        elt = T1[(e - 2)*:32];
        elt = ROR(elt, 17) XOR ROR(elt, 19) XOR LSR(elt, 10);
        elt = elt + x[e*:32] + T0[e*:32];
        result[e*:32] = elt;
    end;
    Q(d>>1) = result;
end;
```
