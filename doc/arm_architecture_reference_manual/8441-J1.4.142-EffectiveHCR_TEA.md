##### J1.4.142 EffectiveHCR\_TEA

```
// EffectiveHCR_TEA()
// ==================
// Return the Effective value of HCR_EL2.TEA.
func EffectiveHCR_TEA() => bit
begin
    if EL2Enabled() && IsFeatureImplemented(FEAT_RAS) then
        return (if ELUsingAArch32(EL2) then HCR2().TEA else HCR_EL2().TEA);
    else
        return ‘0’;
    end;
end;
```
