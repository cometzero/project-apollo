// ResetSVEState()

func ResetSVEState()
begin
    for n = 0 to 31 do
        _Z[[n]] = Zeros{MAX_VL};
    end;
    for n = 0 to 15 do
        _P[[n]] = Zeros{MAX_PL};
    end;
    _FFR = Zeros{MAX_PL};
    FPSR() = ZeroExtend{64}(0x0800009f[31:0]);
    FPMR() = Zeros{64};
end;
