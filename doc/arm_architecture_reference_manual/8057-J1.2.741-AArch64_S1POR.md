##### J1.2.741 AArch64\_S1POR

```
// AArch64_S1POR()
// ===============
// Identify stage 1 permissions overlay register for the acting translation regime
func AArch64_S1POR(regime : Regime) => S1PORType
begin
    if (HaveEL(EL3) && SCR_EL3().PIEn == ‘0’ && regime != Regime_EL3 &&
          ImpDefBool(“SCR_EL3.PIEn forces PIR_ELx or POR_ELx to zero”)) then
        return Zeros{64} as S1PORType;
    end;
    case regime of
        when Regime_EL3 =>  return POR_EL3() as S1PORType;
        when Regime_EL2 =>  return POR_EL2() as S1PORType;
        when Regime_EL20 => return POR_EL2() as S1PORType;
        when Regime_EL10 => return POR_EL1() as S1PORType;
    end;
end;
```
