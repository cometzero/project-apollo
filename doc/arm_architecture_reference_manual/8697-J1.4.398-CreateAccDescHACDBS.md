##### J1.4.398 CreateAccDescHACDBS

// CreateAccDescHACDBS()
// =====================
// Access descriptor for memory accesses to the HACDBS structure.
func CreateAccDescHACDBS() => AccessDescriptor
begin
    var accdesc : AccessDescriptor = NewAccDesc(AccessType_HACDBS);
    accdesc.read            = TRUE;
    return accdesc;
end;
