##### J1.4.99 ExtRegZeroPMUCounters

```
// ExtRegZeroPMUCounters()
// =========================
// Zero set of counters specified by the mask in ‘val’, for external PMU access.
func ExtRegZeroPMUCounters(val : bits(64), addrdesc : AddressDescriptor)
begin
    let masked_val : bits(64) = val AND ExtRegGetPMUAccessMask(addrdesc);
    ApplyPMUZeroMask(masked_val);
    return;
end;
```
