##### J1.3.126 H

```
// H - accessor
// ============
accessor H(n : integer) <=> value : bits(16)
begin
    getter
        assert n >= 0 && n <= 31;
        return S(n)[15:0];
    end;
    setter
        S(n) = ZeroExtend{32}(value);
    end;
end;
```
