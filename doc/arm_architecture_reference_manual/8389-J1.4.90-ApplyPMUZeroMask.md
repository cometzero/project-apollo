##### J1.4.90 ApplyPMUZeroMask

```
// ApplyPMUZeroMask()
// ==================
// Zero the set of counters specified by the mask in ‘masked_val’.
// ‘masked_val’ is expected to already be masked for the permitted counters for the current
// access type before this function is called.
func ApplyPMUZeroMask(masked_val : bits(64))
begin
    for idx = 0 to 63 do
        if masked_val[idx] == ‘1’ then
            case idx of
                when INSTRUCTION_COUNTER_ID =>
                    PMICNTR_EL0() = Zeros{64};
                when CYCLE_COUNTER_ID =>
                    if !HaveAArch64() then
                        PMCCNTR() = Zeros{64};
                    else
                        PMCCNTR_EL0() = Zeros{64};
                    end;
                otherwise =>
                    if !HaveAArch64() then
                        PMEVCNTR(idx) = Zeros{32};
                    elsif IsFeatureImplemented(FEAT_PMUv3p5) then
                        PMEVCNTR_EL0(idx) = Zeros{64};
                    else
                        PMEVCNTR_EL0(idx)[31:0] = Zeros{32};
                    end;
            end;
        end;
    end;
    return;
end;
```
