// EffectiveSCR_EL3_TCR2En()
// =========================
// Returns the Effective value of SCR_EL3().TCR2En.
readonly func EffectiveSCR_EL3_TCR2En() => bit
begin
    assert !ELUsingAArch32(EL3);

end;
