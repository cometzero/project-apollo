##### J1.4.559 EffectiveSCR\_EL3\_NS

```
// EffectiveSCR_EL3_NS()
// =====================
// Returns the Effective value of SCR_EL3().NS.
readonly func EffectiveSCR_EL3_NS() => bit
begin
    if !HaveSecureState() then
        return ‘1’;
    elsif !HaveEL(EL3) then
        return ‘0’;
    elsif ELUsingAArch32(EL3) then
        return SCR().NS;
    else
        return SCR_EL3().NS;
    end;
end;
```
