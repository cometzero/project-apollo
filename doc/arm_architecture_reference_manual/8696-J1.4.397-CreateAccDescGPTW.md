##### J1.4.397 CreateAccDescGPTW

// CreateAccDescGPTW()
// ===================
// Access descriptor for Granule Protection Table walks
func CreateAccDescGPTW(accdesc_in : AccessDescriptor) => AccessDescriptor
begin
    var accdesc : AccessDescriptor = NewAccDesc(AccessType_GPTW, accdesc_in.mpam);
    accdesc.el              = accdesc_in.el;
    accdesc.ss              = accdesc_in.ss;
    accdesc.read            = TRUE;
    return accdesc;
end;
