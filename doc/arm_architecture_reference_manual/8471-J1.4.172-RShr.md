##### J1.4.172 RShr

```
// RShr()
// ======
// Shift integer value right with rounding
func RShr(value : integer, shift : integer, round : boolean) => integer
begin
    assert shift > 0;
    if round then
        return (value + (1 << (shift - 1))) >> shift;
    else
        return value >> shift;
    end;
end;
```
