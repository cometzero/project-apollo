##### J1.4.149 ExceptionSyndrome

```
// ExceptionSyndrome()
// ===================
// Return a blank exception syndrome record for an exception of the given type.
func ExceptionSyndrome(exceptype : Exception) => ExceptionRecord
begin
    var r : ExceptionRecord;
    r.exceptype      = exceptype;
    // Initialize all other fields
    r.syndrome.iss   = Zeros{25};
    r.syndrome.iss2  = Zeros{24};
    r.vaddress       = Zeros{64};
    r.ipavalid       = FALSE;
    r.pavalid        = FALSE;
    r.NS             = ‘0’;
    r.ipaddress      = Zeros{NUM_PABITS};
    r.paddress.paspace = ARBITRARY : PASpace;
    r.paddress.address = ARBITRARY : bits(NUM_PABITS);
    r.trappedsyscallinst = FALSE;
    return r;
end;
```
