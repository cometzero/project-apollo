##### J1.2.423 CheckEnADERRApplies

```
// CheckEnADERRApplies()
// =====================
// Returns TRUE if SCTLR2_ELx().EnADERR applies to the given memory read access.
func CheckEnADERRApplies(accdesc : AccessDescriptor) => boolean
begin
     var imp_specific_access : boolean = FALSE;
    case accdesc.acctype of
        when AccessType_ASIMD  => imp_specific_access = TRUE;
        when AccessType_SVE    => imp_specific_access = TRUE;
        when AccessType_SME    => imp_specific_access = TRUE;
        when AccessType_GPR    =>
            imp_specific_access = FALSE;
            imp_specific_access = imp_specific_access || accdesc.atomicop;
            imp_specific_access = imp_specific_access || accdesc.rcw || accdesc.rcws;
            imp_specific_access = imp_specific_access || (accdesc.ls64 && accdesc.withstatus);
        when AccessType_NV2    => imp_specific_access = TRUE;
        otherwise              => imp_specific_access = FALSE;
    end;
    if imp_specific_access then
        return ImpDefBool(“EnADERR applies to given memory read AccessType”);
    end;
    return TRUE;
end;
```
