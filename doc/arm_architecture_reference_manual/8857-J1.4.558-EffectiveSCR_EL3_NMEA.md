##### J1.4.558 EffectiveSCR\_EL3\_NMEA

```
// EffectiveSCR_EL3_NMEA()
// =======================
// Returns the Effective value of SCR_EL3().NMEA.
readonly func EffectiveSCR_EL3_NMEA() => bit
begin
    assert !ELUsingAArch32(EL3);
    assert IsFeatureImplemented(FEAT_DoubleFault);
    if !HaveEL(EL3) then return ‘0’; end;
    if !IsFeatureImplemented(FEAT_DoubleFault2) && EffectiveSCR_EL3_EA() == ‘0’ then
        return ‘0’;
    end;
    return SCR_EL3().NMEA;
end;
```
