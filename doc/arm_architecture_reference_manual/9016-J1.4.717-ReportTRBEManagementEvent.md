##### J1.4.717 ReportTRBEManagementEvent

```
// ReportTRBEManagementEvent()
// ===========================
// Report a buffer management event with the event class ‘ec’ and status code ‘bsc’
func ReportTRBEManagementEvent(ec : bits(6), bsc : bits(6))
begin
    let target_el : bits(2) = DefaultTRBEEvent();
    if TRBSR_EL(target_el).S == ‘0’ then
        TRBSR_EL(target_el).S    = ‘1’;     // Stop collection
        if TRBSR_EL(target_el).IRQ == ‘0’ then
            TRBSR_EL(target_el).IRQ  = ‘1’;     // Assert interrupt or exception
            CTI_SignalEvent(CrossTriggerIn_TRBEMgmt);
        end;
        TRBSR_EL(target_el).EC   = ec;
        TRBSR_EL(target_el).MSS  = ZeroExtend{16}(bsc);
        TRBSR_EL(target_el).MSS2 = Zeros{24};
        CTI_SignalEvent(CrossTriggerIn_TRBEStop);
    end;
end;
```
