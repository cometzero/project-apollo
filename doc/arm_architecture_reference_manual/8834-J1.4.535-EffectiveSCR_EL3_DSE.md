##### J1.4.535 EffectiveSCR\_EL3\_DSE

```
// EffectiveSCR_EL3_DSE()
// ======================
// Returns the Effective value of SCR_EL3().DSE.
readonly func EffectiveSCR_EL3_DSE() => bit
begin
    assert !ELUsingAArch32(EL3);
    assert IsFeatureImplemented(FEAT_E3DSE);
    if !HaveEL(EL3) then return ‘0’; end;
    if EffectiveSCR_EL3_EnDSE() == ‘0’ then return ‘0’; end;
    return SCR_EL3().DSE;
end;
```
