##### J1.4.551 EffectiveSCR\_EL3\_GPF

```
// EffectiveSCR_EL3_GPF()
// ======================
// Returns the Effective value of SCR_EL3().GPF.
readonly func EffectiveSCR_EL3_GPF() => bit
begin
    assert IsFeatureImplemented(FEAT_RME);
    if !HaveEL(EL3) then return ‘0’; end;
    return SCR_EL3().GPF;
end;
```
