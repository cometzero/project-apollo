##### J1.4.252 BFInfinity

```
// BFInfinity()
// ============
func BFInfinity{N}(sign : bit) => bits(N)
begin
    assert N == 16;
    let E : integer{} = 8;
    let F : integer{} = N - (E + 1);
    return sign :: Ones{E} :: Zeros{F};
end;
```
