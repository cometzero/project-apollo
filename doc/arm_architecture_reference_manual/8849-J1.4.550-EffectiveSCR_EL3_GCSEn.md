##### J1.4.550 EffectiveSCR\_EL3\_GCSEn

```
// EffectiveSCR_EL3_GCSEn()
// ========================
// Returns the Effective value of SCR_EL3().GCSEn.
readonly func EffectiveSCR_EL3_GCSEn() => bit
begin
    assert IsFeatureImplemented(FEAT_GCS);
    if !HaveEL(EL3) then return ‘1’; end;
    return SCR_EL3().GCSEn;
end;
```
