##### J1.4.182 TestEventCNTP

```
// TestEventCNTP()
// ===============
// Generate Event stream from the physical counter
func TestEventCNTP(prev_physical_count : bits(64), current_physical_count : bits(64))
begin
    var n : integer;
    if CNTHCTL_EL2().EVNTEN == ‘1’ then
        n = UInt(CNTHCTL_EL2().EVNTI);
        if IsFeatureImplemented(FEAT_ECV) && CNTHCTL_EL2().EVNTIS == ‘1’ then
            n = n + 8;
        end;
        if CNTHCTL_EL2().EVNTDIR == ‘0’ then
            if prev_physical_count[n] == ‘0’ && current_physical_count[n] == ‘1’ then
                SetEventRegister();
            end;
        else
            if prev_physical_count[n] == ‘1’ && current_physical_count[n] == ‘0’ then
                SetEventRegister();
            end;
        end;
    end;
    return;
end;
```
