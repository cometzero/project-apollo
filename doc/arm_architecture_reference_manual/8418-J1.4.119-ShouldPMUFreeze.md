##### J1.4.119 ShouldPMUFreeze

```
// ShouldPMUFreeze()
// =================
func ShouldPMUFreeze(r : PMUCounterRange) => boolean
begin
    let include_r1 : boolean = (r == PMUCounterRange_R1);
    let include_r2 : boolean = (r == PMUCounterRange_R2);
    let include_r3 : boolean = FALSE;
    if r == PMUCounterRange_R3 then
        return FALSE;
    end;
    let overflow : boolean = CheckPMUOverflowCondition(PMUOverflowCondition_Freeze,
                                                       include_r1, include_r2, include_r3);
    return overflow;
end;
```
