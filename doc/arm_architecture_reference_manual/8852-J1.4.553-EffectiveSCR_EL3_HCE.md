##### J1.4.553 EffectiveSCR\_EL3\_HCE

```
// EffectiveSCR_EL3_HCE()
// ======================
// Returns the Effective value of SCR_EL3().HCE.
readonly func EffectiveSCR_EL3_HCE() => bit
begin
    assert !ELUsingAArch32(EL3);
    if !HaveEL(EL2) then return ‘0’; end;
    if !HaveEL(EL3) then return ‘1’; end;
    return SCR_EL3().HCE;
end;
```
