##### J1.2.579 AArch64\_RandomTag

```
// AArch64_RandomTag()
// ===================
// Generate a random Allocation Tag.
func AArch64_RandomTag() => bits(4)
begin
    var tag : bits(4);
    for i = 0 to 3 do
        tag[i] = AArch64_NextRandomTagBit();
    end;
    return tag;
end;
```
