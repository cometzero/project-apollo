##### J1.4.532 EffectiveSCR\_EL3\_APK

```
// EffectiveSCR_EL3_APK()
// ======================
// Returns the Effective value of SCR_EL3().APK.
readonly func EffectiveSCR_EL3_APK() => bit
begin
    assert !ELUsingAArch32(EL3);
    assert IsFeatureImplemented(FEAT_PAuth);
    if !HaveEL(EL3) then return ‘1’; end;
    return SCR_EL3().APK;
end;
```
