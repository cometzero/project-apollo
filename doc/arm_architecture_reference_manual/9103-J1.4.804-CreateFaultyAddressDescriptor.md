##### J1.4.804 CreateFaultyAddressDescriptor

```
// CreateFaultyAddressDescriptor()
// ===============================
// Set internal members for address descriptor type with values indicating error
func CreateFaultyAddressDescriptor(fault : FaultRecord) => AddressDescriptor
begin
    var addrdesc : AddressDescriptor;
    addrdesc.vaddress = fault.vaddress;
    addrdesc.fault    = fault;
    return addrdesc;
end;
```
