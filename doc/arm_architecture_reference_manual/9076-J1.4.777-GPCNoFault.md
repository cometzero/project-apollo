##### J1.4.777 GPCNoFault

```
// GPCNoFault()
// ============
// Returns the default properties of a GPCF that does not represent a fault
func GPCNoFault() => GPCFRecord
begin
    var result : GPCFRecord;
    result.gpf = GPCF_None;
    return result;
end;
```
