##### J1.4.818 Stage2OA

```
// Stage2OA()
// ==========
// Given the final walk state (a page or block descriptor), map the untranslated
// IPA bits to the output PA.
func Stage2OA(ipa : bits(NUM_PABITS), walkparams : S2TTWParams, walkstate : TTWState) => FullAddress
begin
    // Output Address
    var oa : FullAddress;
    let tsize : integer = TranslationSize(walkparams.d128, walkparams.tgx, walkstate.level);
    var csize : integer;
    if walkstate.contiguous == ‘1’ then
        csize = ContiguousSize(walkparams.d128, walkparams.tgx, walkstate);
    else
        csize = 0;
    end;
    let ipa_msb : AddressSize = (tsize + csize) as AddressSize;
    oa.paspace = walkstate.baseaddress.paspace;
    oa.address = walkstate.baseaddress.address[NUM_PABITS-1:ipa_msb]::ipa[ipa_msb-1:0];
    return oa;
end;
```
