##### J1.2.437 IsNormalMemoryReadErrorSynchronous

```
// IsNormalMemoryReadErrorSynchronous()
// ====================================
// Returns TRUE when error signaled on a load from Normal memory is taken synchronously.
func IsNormalMemoryReadErrorSynchronous(accdesc : AccessDescriptor, desc : AddressDescriptor)
                                          => boolean
begin
    if IsFeatureImplemented(FEAT_ANERR) && CheckEnANERRApplies(accdesc, desc) then
        case S1TranslationRegime(accdesc.el) of
            when EL3 =>
                if ((!IsFeatureImplemented(FEAT_ADERR) ||
                      GetANERRErrorMode() != ErrSignalMode_TiedToPeer ||
                      SCTLR2_EL3().EnADERR == ‘0’) && SCTLR2_EL3().EnANERR == ‘0’) then
                    return TRUE;
                end;
            when EL2 =>
                if (!IsSCTLR2EL2Enabled() ||
                      ((!IsFeatureImplemented(FEAT_ADERR) ||
                        GetANERRErrorMode() != ErrSignalMode_TiedToPeer ||
                        SCTLR2_EL2().EnADERR == ‘0’) && SCTLR2_EL2().EnANERR == ‘0’)) then
                    return TRUE;
                end;
            when EL1 =>
                if EffectiveHCRX_EL2_EnSNERR() == ‘1’ then
                    return TRUE;
                end;
                if (!IsSCTLR2EL1Enabled() ||
                      ((!IsFeatureImplemented(FEAT_ADERR) ||
                        GetANERRErrorMode() != ErrSignalMode_TiedToPeer ||
                        SCTLR2_EL1().EnADERR == ‘0’) &&
                       SCTLR2_EL1().EnANERR == ‘0’)) then
                    return TRUE;
                end;
            otherwise => unreachable;
        end;
    end;
    return ImpDefBool(“Normal read error sync when EnANERR not implemented or enabled”);
end;
```
