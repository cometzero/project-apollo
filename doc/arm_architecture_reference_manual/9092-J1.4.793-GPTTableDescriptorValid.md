##### J1.4.793 GPTTableDescriptorValid

```
// GPTTableDescriptorValid()
// =========================
// Returns TRUE if the given GPT Table descriptor is valid, and FALSE otherwise.
func GPTTableDescriptorValid(level_0_entry : bits(64)) => boolean
begin
    assert level_0_entry[3:0] == GPT_Table;
    let l0sz : integer{} = GPTL0Size();
    let pgs : PGSe       = DecodePGS(GPCCR_EL3().PGS);
    let p : integer{}    = DecodePGSRange(pgs);
    let top : integer{}  = if DecodePPS() == 56 then 56 else 52;
    return IsZero(level_0_entry[63:top,11:4]) && IsZero(level_0_entry[(l0sz-p)-2:12]);
end;
```
