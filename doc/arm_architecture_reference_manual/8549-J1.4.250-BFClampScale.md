##### J1.4.250 BFClampScale

```
// BFClampScale()
// ==============
func BFClampScale(op : bits(16), scale_in : integer) => integer
begin
    let E : integer = 8;
    let F : integer = 7;
    let exp : integer = UInt(op[14:7]);
    let emax : integer = (1 << E) - 1;
    let min_scale : integer = -(F + 1);
    let max_scale : integer = emax + (F + 1);
    let scale : integer = Max(min_scale - exp, Min(scale_in, max_scale - exp));
    return scale;
end;
```
