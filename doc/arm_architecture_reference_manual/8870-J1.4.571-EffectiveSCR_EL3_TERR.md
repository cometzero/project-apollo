##### J1.4.571 EffectiveSCR\_EL3\_TERR

```
// EffectiveSCR_EL3_TERR()
// =======================
// Returns the Effective value of SCR_EL3().TERR.
readonly func EffectiveSCR_EL3_TERR() => bit
begin
    assert !ELUsingAArch32(EL3);
    assert IsFeatureImplemented(FEAT_RAS);
    if !HaveEL(EL3) then return ‘0’; end;
    return SCR_EL3().TERR;
end;
```
