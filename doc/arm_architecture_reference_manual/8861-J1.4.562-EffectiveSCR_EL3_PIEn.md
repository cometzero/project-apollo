##### J1.4.562 EffectiveSCR\_EL3\_PIEn

```
// EffectiveSCR_EL3_PIEn()
// =======================
// Returns the Effective value of SCR_EL3().PIEn.
readonly func EffectiveSCR_EL3_PIEn() => bit
begin
    assert !ELUsingAArch32(EL3);
    assert (IsFeatureImplemented(FEAT_S1PIE) || IsFeatureImplemented(FEAT_S2PIE) ||
            IsFeatureImplemented(FEAT_S1POE) || IsFeatureImplemented(FEAT_S2POE));
    if !HaveEL(EL3) then return ‘1’; end;
    return SCR_EL3().PIEn;
end;
```
