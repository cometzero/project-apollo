##### J1.2.587 BTypeCompatible\_PACIXSP

```
// BTypeCompatible_PACIXSP()
// =========================
// Returns TRUE if PACIASP or PACIBSP instructions are implicitly compatible with PSTATE.BTYPE,
// FALSE otherwise.
func BTypeCompatible_PACIXSP() => boolean
begin
    let pacinst : PACInstType = PACIxSP;
    return BTypeCompatible_PAC(pacinst);
end;
```
