##### J1.4.84 CheckSoftwareAccessToDebugRegisters

```
// CheckSoftwareAccessToDebugRegisters()
// =====================================
// Check for access to Breakpoint and Watchpoint registers.
func CheckSoftwareAccessToDebugRegisters()
begin
    let os_lock : bit = (if ELUsingAArch32(EL1) then DBGOSLSR().OSLK else OSLSR_EL1().OSLK);
    if HaltingAllowed() && EDSCR().TDA == ‘1’ && os_lock == ‘0’ then
        Halt(DebugHalt_SoftwareAccess);
    end;
end;
```
