# ​J1.2.559 PFAR_EL

Source: <https://developer.arm.com/documentation/ddi0487/mc/-Part-J-Architectural-Pseudocode/-Chapter-J1-A-profile-Architecture-Pseudocode/-J1-2-Pseudocode-for-AArch64-operation/-J1-2-559-PFAR-EL>

##### J1.2.559 PFAR\_EL

```
// PFAR_EL - accessor
// ==================
accessor PFAR_EL(regime : bits(2)) <=> value : bits(64)
begin
    getter
        assert (IsFeatureImplemented(FEAT_PFAR) ||
                 (regime == EL3 && IsFeatureImplemented(FEAT_RME)));
        var r : bits(64);
        case regime of
            when EL1 => r = PFAR_EL1();
            when EL2 => r = PFAR_EL2();
            when EL3 => r = MFAR_EL3();
            otherwise => unreachable;
        end;
        return r;
    end;
    setter
        let r : bits(64) = value;
        assert (IsFeatureImplemented(FEAT_PFAR) ||
                   (IsFeatureImplemented(FEAT_RME) && regime == EL3));
        case regime of
            when EL1 => PFAR_EL1() = r;
            when EL2 => PFAR_EL2() = r;
            when EL3 => MFAR_EL3() = r;
            otherwise => unreachable;
        end;
    end;
end;
```
