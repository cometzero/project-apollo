##### J1.3.169 AArch32\_EffectiveBroadcast

```
// AArch32_EffectiveBroadcast()
// ============================
func AArch32_EffectiveBroadcast(broadcast_in : Broadcast) => Broadcast
begin
    var broadcast : Broadcast = broadcast_in;
    if PSTATE.EL != EL1 then return broadcast; end;
    if (broadcast == Broadcast_NSH && ((ELUsingAArch32(EL2) && HCR().FB == ‘1’) ||
          (!ELUsingAArch32(EL2) && HCR_EL2().FB == ‘1’))) then
        broadcast = Broadcast_ForcedISH;
    end;
    return broadcast;
end;
```
