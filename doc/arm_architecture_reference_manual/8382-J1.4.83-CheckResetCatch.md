##### J1.4.83 CheckResetCatch

```
// CheckResetCatch()
// =================
// Called after reset
func CheckResetCatch()
begin
    if ((IsFeatureImplemented(FEAT_DoPD) && CTIDEVCTL().RCE == ‘1’) ||
          (!IsFeatureImplemented(FEAT_DoPD) && EDECR().RCE == ‘1’)) then
        EDESR().RC = ‘1’;
        // If halting is allowed then halt immediately
        if HaltingAllowed() then Halt(DebugHalt_ResetCatch); end;
    end;
end;
```
