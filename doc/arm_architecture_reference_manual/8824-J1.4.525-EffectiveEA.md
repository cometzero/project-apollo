##### J1.4.525 EffectiveEA

```
// EffectiveEA()
// =============
// Returns the Effective value of SCR_EL3().EA
readonly func EffectiveEA() => bit
begin
    if !HaveEL(EL3) || Halted() then
        return ‘0’;
    else
        return if HaveAArch64() then SCR_EL3().EA else SCR().EA;
    end;
end;
```
