##### J1.2.261 GCSPUSHX

```
// GCSPUSHX()
// ==========
// Called to push a GCS exception return record.
func GCSPUSHX()
begin
    let spsr : bits(64) = SPSR_ELx();
    AddGCSExRecord(ELR_ELx(), spsr, X{64}(30));
    PSTATE.EXLOCK = ‘0’;
    return;
end;
```
