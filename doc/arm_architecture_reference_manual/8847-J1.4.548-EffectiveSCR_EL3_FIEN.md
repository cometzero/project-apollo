##### J1.4.548 EffectiveSCR\_EL3\_FIEN

```
// EffectiveSCR_EL3_FIEN()
// =======================
// Returns the Effective value of SCR_EL3().FIEN.
readonly func EffectiveSCR_EL3_FIEN() => bit
begin
    assert !ELUsingAArch32(EL3);
    assert IsFeatureImplemented(FEAT_RASv1p1);
    if !HaveEL(EL3) then return ‘1’; end;
    return SCR_EL3().FIEN;
end;
```
