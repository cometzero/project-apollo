##### J1.4.121 SysRegZeroCycleCounter

```
// SysRegZeroCycleCounter()
// ========================
// Called on a System register write to PMCR_EL0 or PMCR that writes ‘1’ to PMCR_EL0.C or PMCR.C.
func SysRegZeroCycleCounter()
begin
    var mask : bits(64) = Zeros{};
    mask[CYCLE_COUNTER_ID] = ‘1’;
    SysRegZeroPMUCounters(mask);
end;
```
