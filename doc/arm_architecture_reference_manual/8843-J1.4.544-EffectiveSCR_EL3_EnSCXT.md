##### J1.4.544 EffectiveSCR\_EL3\_EnSCXT

```
// EffectiveSCR_EL3_EnSCXT()
// =========================
// Returns the Effective value of SCR_EL3().EnSCXT.
readonly func EffectiveSCR_EL3_EnSCXT() => bit
begin
    assert !ELUsingAArch32(EL3);
    assert (IsFeatureImplemented(FEAT_CSV2_2) || IsFeatureImplemented(FEAT_CSV2_1p2));
    if !HaveEL(EL3) then return ‘1’; end;
    return SCR_EL3().EnSCXT;
end;
```
