##### J1.2.433 IsDeviceMemoryReadErrorSynchronous

```
// IsDeviceMemoryReadErrorSynchronous()
// ====================================
// Returns TRUE when the Device memory read is synchronous in the current Exception level.
func IsDeviceMemoryReadErrorSynchronous(accdesc : AccessDescriptor, desc : AddressDescriptor)
                                           => boolean
begin
    if IsFeatureImplemented(FEAT_ADERR) && CheckEnADERRApplies(accdesc) then
        case S1TranslationRegime(accdesc.el) of
            when EL3 =>
                if HaveEL(EL3) && SCR_EL3().SCTLR2En == ‘0’ then
                    return TRUE;
                end;
                if ((!IsFeatureImplemented(FEAT_ANERR) ||
                     GetADERRErrorMode() != ErrSignalMode_TiedToPeer ||
                     SCTLR2_EL3().EnANERR == ‘0’) && SCTLR2_EL3().EnADERR == ‘0’) then
                    return TRUE;
                end;
            when EL2 =>
                if (!IsSCTLR2EL2Enabled() ||
                      ((!IsFeatureImplemented(FEAT_ANERR) ||
                        GetADERRErrorMode() != ErrSignalMode_TiedToPeer ||
                        SCTLR2_EL2().EnANERR == ‘0’) && SCTLR2_EL2().EnADERR == ‘0’)) then
                    return TRUE;
                end;
            when EL1 =>
                if IsHCRXEL2Enabled() && EffectiveHCRX_EL2_EnSDERR() == ‘1’ then
                    return TRUE;
                end;
                if (!IsSCTLR2EL1Enabled() ||
                      ((!IsFeatureImplemented(FEAT_ANERR) ||
                        GetADERRErrorMode() != ErrSignalMode_TiedToPeer ||
                        SCTLR2_EL1().EnANERR == ‘0’) &&
                       SCTLR2_EL1().EnADERR == ‘0’)) then
                    return TRUE;
                end;
            otherwise => unreachable;
        end;
    end;
    return ImpDefBool(“Device read error sync when EnADERR not implemented or enabled”);
end;
```
