##### J1.4.566 EffectiveSCR\_EL3\_SIF

```
// EffectiveSCR_EL3_SIF()
// ======================
// Returns the Effective value of SCR_EL3().SIF.
readonly func EffectiveSCR_EL3_SIF() => bit
begin
    assert !ELUsingAArch32(EL3);
    if !HaveEL(EL3) then return ‘0’; end;
    if IsFeatureImplemented(FEAT_RME) && !IsFeatureImplemented(FEAT_SEL2) then
        return ‘0’;
    end;
    return SCR_EL3().SIF;
end;
```
