// FAR_EL - accessor
// =================
accessor FAR_EL(regime : bits(2)) <=> value : bits(64)
begin
    getter
        var r : bits(64);
        case regime of
            when EL1 =>  r = FAR_EL1();
            when EL2 =>  r = FAR_EL2();
            when EL3 =>  r = FAR_EL3();
            otherwise => unreachable;
        end;
        return r;
    end;
    setter
        let r : bits(64) = value;
        case regime of
            when EL1 =>  FAR_EL1() = r;
            when EL2 =>  FAR_EL2() = r;
            when EL3 =>  FAR_EL3() = r;
            otherwise => unreachable;
        end;
    end;
end;
