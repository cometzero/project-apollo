##### J1.2.424 CheckEnANERRApplies

```
// CheckEnANERRApplies()
// =====================
// Returns TRUE if SCTLR2_ELx().EnANERR applies to the given memory read access.
func CheckEnANERRApplies(accdesc : AccessDescriptor, desc : AddressDescriptor) => boolean
begin
    var imp_specific_access : boolean = FALSE;
    case accdesc.acctype of
        when AccessType_GPR =>
            imp_specific_access = FALSE;
            imp_specific_access = imp_specific_access || accdesc.atomicop;
            imp_specific_access = imp_specific_access || accdesc.rcw || accdesc.rcws;
            imp_specific_access = imp_specific_access || (accdesc.ls64 && accdesc.withstatus);
        when {AccessType_ASIMD, AccessType_SVE} =>
            imp_specific_access = IsFeatureImplemented(FEAT_SME) && PSTATE.SM == ‘1’;
        when AccessType_SME =>
            imp_specific_access = TRUE;
        when AccessType_NV2 =>
            imp_specific_access = !IsWBShareable(desc.memattrs);
        otherwise =>
            imp_specific_access = FALSE;
    end;
    if imp_specific_access then
        return ImpDefBool(“EnANERR applies to given memory read AccessType”);
    end;
    return TRUE;
end;
```
