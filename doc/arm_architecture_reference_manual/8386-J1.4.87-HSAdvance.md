##### J1.4.87 HSAdvance

```
// HSAdvance()
// ===========
// Advance the Halting Step State Machine
func HSAdvance()
begin
    if !ShouldAdvanceHS then return; end;
    let step_enabled : boolean = EDECR().SS == ‘1’ && HaltingAllowed();
    let active_not_pending : boolean = step_enabled && EDESR().SS == ‘0’;
    if active_not_pending then EDESR().SS = ‘1’; end;  // set as pending.
    ShouldAdvanceHS = FALSE;
    return;
end;
```
