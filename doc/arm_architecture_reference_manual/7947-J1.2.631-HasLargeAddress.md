# ​J1.2.631 HasLargeAddress

##### J1.2.631 HasLargeAddress

```
// HasLargeAddress()
// =================
// Returns TRUE if the regime is configured for 52 bit addresses, FALSE otherwise.
func HasLargeAddress(regime : Regime) => boolean
begin
    if !IsFeatureImplemented(FEAT_LPA2) then
        return FALSE;
    end;
    case regime of
        when Regime_EL3 =>
            return TCR_EL3().DS == ‘1’;
        when Regime_EL2 =>
            return TCR_EL2().DS == ‘1’;
        when Regime_EL20 =>
            return TCR_EL2().DS == ‘1’;
        when Regime_EL10 =>
            return TCR_EL1().DS == ‘1’;
        otherwise =>
            unreachable;
    end;
end;
```
