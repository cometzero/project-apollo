##### J1.4.575 EffectiveSCR\_EL3\_TMEA

```
// EffectiveSCR_EL3_TMEA()
// =======================
// Returns the Effective value of SCR_EL3().TMEA
readonly func EffectiveSCR_EL3_TMEA() => bit
begin
    if (IsFeatureImplemented(FEAT_DoubleFault2) && HaveEL(EL3) &&
          !ELUsingAArch32(EL3)) then
        return SCR_EL3().TMEA;
    else
        return ‘0’;
    end;
end;
```
