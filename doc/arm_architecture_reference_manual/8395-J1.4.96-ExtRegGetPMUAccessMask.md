# ​J1.4.96 ExtRegGetPMUAccessMask

##### J1.4.96 ExtRegGetPMUAccessMask

```
// ExtRegGetPMUAccessMask()
// ========================
// Return a mask of the PMU counters accessible via external PMU access.
func ExtRegGetPMUAccessMask(addrdesc : AddressDescriptor) => bits(64)
begin
    var mask : bits(64) = Zeros{};
    if IsFeatureImplemented(FEAT_PMUv3_ICNTR) then
        mask[INSTRUCTION_COUNTER_ID] = ‘1’;
    end;
    mask[CYCLE_COUNTER_ID] = ‘1’;
    let counters : integer{} = (if (IsFeatureImplemented(FEAT_PMUv3_EXTPMN) &&
                                     !IsMostSecureAccess(addrdesc)) then UInt(EffectiveEPMN())
                                else NUM_PMU_COUNTERS);
    if counters > 0 then
        mask[counters-1:0] = Ones{counters};
    end;
    return mask;
end;
```
