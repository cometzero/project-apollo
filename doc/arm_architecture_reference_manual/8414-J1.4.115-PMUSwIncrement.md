##### J1.4.115 PMUSwIncrement

```
// PMUSwIncrement()
// ================
// Generate PMU Events on a System register write to PMSWINC
func PMUSwIncrement(sw_incr_in : bits(64))
begin
    var sw_incr : bits(64)   = sw_incr_in;
    var mask : bits(31)      = Zeros{};
    let counters : integer{} = GetNumEventCountersAccessible();
    if counters > 0 then
        mask[counters-1:0] = Ones{counters};
    end;
    if (IsFeatureImplemented(FEAT_PMUv3p9) && !ELUsingAArch32(EL1) &&
          PSTATE.EL == EL0 && PMUSERENR_EL0().[UEN,SW] == ‘10’) then
        mask = mask AND PMUACR_EL1()[30:0];
    end;
    sw_incr = sw_incr AND ZeroExtend{64}(mask);
    for idx = 0 to 30 do
        if sw_incr[idx] == ‘1’ then
            PMUEvent(PMU_EVENT_SW_INCR, 1, idx);
        end;
    end;
    return;
end;
```
