##### J1.2.622 AArch64\_TLBI\_RVA

```
// AArch64_TLBI_RVA()
// ==================
// Range invalidate by VA range all stage 1 TLB entries in the indicated
// broadcast domain matching the indicated VMID and ASID (where regime
// supports VMID, ASID) in the indicated regime with the indicated security state.
// ASID, and range related parameters are derived from Xt.
// Note: stage 1 and stage 2 combined entries are in the scope of this operation.
func AArch64_TLBI_RVA(security : SecurityState, regime_in : Regime, vmid_in : bits(NUM_VMIDBITS),
                      broadcast_in : Broadcast, level : TLBILevel, attr_in : TLBIMemAttr,
                      Xt : bits(64))
begin
    assert PSTATE.EL IN {EL3, EL2, EL1};
    var broadcast : Broadcast   = AArch64_EffectiveBroadcast(broadcast_in);
    var attr      : TLBIMemAttr = attr_in;
    if attr == TLBI_AllAttr && ExcludeXS() then
        attr = TLBI_ExcludeXS;
    end;
    var regime : Regime   = regime_in;
    var vmid   : bits(NUM_VMIDBITS) = vmid_in;
    if regime == Regime_EL10 && ELIsInHost(EL0) then
        regime = Regime_EL20;
        vmid   = VMID_NONE;
    elsif regime == Regime_EL2 && ELIsInHost(EL2) then
        regime = Regime_EL20;
    end;
    var r : TLBIRecord;
    r.op           = TLBIOp_RVA;
    r.from_aarch64 = TRUE;
    r.security     = security;
    r.regime       = regime;
    r.vmid         = vmid;
    r.use_vmid     = UseVMID(regime);
    r.level        = level;
    r.attr         = attr;
    r.asid         = Xt[63:48];
    r.ttl[1:0]     = Xt[38:37];
    r.d64          = TRUE;
    r.d128         = r.ttl[1:0] == ‘00’;
    var valid : boolean;
    (valid, r.tg, r.address, r.end_address) = TLBIRange(regime, Xt);
    if !valid then return; end;
    TLBI(r);
    if broadcast != Broadcast_NSH then BroadcastTLBI(broadcast, r); end;
    return;
end;
```
