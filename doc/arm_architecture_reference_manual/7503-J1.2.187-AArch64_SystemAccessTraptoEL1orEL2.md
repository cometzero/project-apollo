# ​J1.2.187 AArch64_SystemAccessTraptoEL1orEL2

Source: <https://developer.arm.com/documentation/ddi0487/mc/-Part-J-Architectural-Pseudocode/-Chapter-J1-A-profile-Architecture-Pseudocode/-J1-2-Pseudocode-for-AArch64-operation/-J1-2-187-AArch64-SystemAccessTraptoEL1orEL2>

##### J1.2.187 AArch64\_SystemAccessTraptoEL1orEL2

```
// AArch64_SystemAccessTraptoEL1orEL2()
// ====================================
// Trapped access to AArch64 System register or system instruction from EL0.
// The trap is taken to EL1, or to EL2 if HCR_EL2().TGE is 1.
func AArch64_SystemAccessTraptoEL1orEL2(ec : integer)
begin
    assert PSTATE.EL == EL0;
    let target_el : bits(2) = if EL2Enabled() && HCR_EL2().TGE == ‘1’ then EL2 else EL1;
    AArch64_SystemAccessTrap(target_el, ec);
end;
```
