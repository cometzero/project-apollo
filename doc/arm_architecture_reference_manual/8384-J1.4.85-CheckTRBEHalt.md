// CheckTRBEHalt()
// ===============
func CheckTRBEHalt()
begin
    if !IsFeatureImplemented(FEAT_Debugv8p9) || !IsFeatureImplemented(FEAT_TRBE_EXT) then
        return;
    end;
    if TraceBufferEnabled() && TRBSR_EL1().IRQ == ‘1’ && EDECR().TRBE == ‘1’ then
        ExternalDebugRequest();
    end;
end;
