// GenMPAMTRBE()
// =============
// Returns MPAMinfo for Trace Buffer Unit in external mode.
func GenMPAMTRBE() => MPAMinfo
begin
    assert !SelfHostedTraceEnabled();
    var pspace : PARTIDSpaceType;
    case TRBMAR_EL1().PAS of
        when ‘00’ => pspace = PIDSpace_Secure;
        when ‘01’ => pspace = PIDSpace_NonSecure;
        when ‘10’ => pspace = PIDSpace_Root;
        when ‘11’ => pspace = PIDSpace_Realm;
    end;
    if !IsFeatureImplemented(FEAT_TRBE_MPAM) || TRBMPAM_EL1().EN == ‘0’ then
        return DefaultMPAMInfo(pspace);
    end;
    assert IsFeatureImplemented(FEAT_TRBE_MPAM);
    var mpaminfo : MPAMinfo;
    mpaminfo.mpam_sp = pspace;
    mpaminfo.partid = TRBMPAM_EL1().PARTID;
    var partid_out_of_range : boolean = FALSE;
    // When the PARTID is out-of-range, the behavior is a CONSTRAINED UNPREDICTABLE choice of:
    //     - Replace the out-of-range PARTID with any in-range PARTID
    //     - Replace the out-of-range PARTID with the default PARTID
    if UInt(mpaminfo.partid) > UInt(MPAMIDR_EL1().PARTID_MAX) then
        partid_out_of_range = TRUE;
        var c : Constraint;
        var partid : integer;
        (c, partid) = ConstrainUnpredictableInteger(0, UInt(MPAMIDR_EL1().PARTID_MAX),
                                                    Unpredictable_OUT_OF_RANGE_ID);
        assert c IN {Constraint_DEFAULT_ID, Constraint_ANY_ID};
        case c of
            when Constraint_DEFAULT_ID => mpaminfo.partid = DEFAULT_PARTID;
            when Constraint_ANY_ID     => mpaminfo.partid = partid[15:0];
        end;
    end;
    if partid_out_of_range && ConstrainUnpredictableBool(Unpredictable_USE_DEFAULT_PMG) then
        mpaminfo.pmg = DEFAULT_PMG;
    else
        mpaminfo.pmg = TRBMPAM_EL1().PMG;
    end;
    if UInt(mpaminfo.pmg) > UInt(MPAMIDR_EL1().PMG_MAX) then
        var c : Constraint;
        var pmg : integer;
        (c, pmg) = ConstrainUnpredictableInteger(0, UInt(MPAMIDR_EL1().PMG_MAX),
                                                 Unpredictable_OUT_OF_RANGE_ID);
        assert c IN {Constraint_DEFAULT_ID, Constraint_ANY_ID};
        case c of
            when Constraint_DEFAULT_ID => mpaminfo.pmg = DEFAULT_PMG;
            when Constraint_ANY_ID     => mpaminfo.pmg = pmg[7:0];
        end;
    end;
    return mpaminfo;
end;
