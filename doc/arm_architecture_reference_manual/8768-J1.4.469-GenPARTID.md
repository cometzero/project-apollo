##### J1.4.469 GenPARTID

```
// GenPARTID()
// ===========
// Returns physical PARTID and error boolean for Exception level el.
// If mpamdata.sm is TRUE then PARTID is from MPAMSM_EL1.
// Otherwise, the PARTID is from MPAMn_ELx.
func GenPARTID(el : bits(2), mpamdata : MPAMdata) => (PARTIDType, boolean)
begin
    var partidel : PARTIDType = GetMPAM_PARTID(el, mpamdata);
    var partid_err : boolean  = FALSE;
    // When the PARTID is out-of-range, the behaviour is a CONSTRAINED UNPREDICTABLE choice of:
    //     - Replace the out-of-range PARTID with any in-range PARTID
    //     - Replace the out-of-range PARTID with the default PARTID
    if UInt(partidel) > UInt(MPAMIDR_EL1().PARTID_MAX) then
        var c : Constraint;
        var partid : integer;
        partid_err = TRUE;
        (c, partid) = ConstrainUnpredictableInteger(0, UInt(MPAMIDR_EL1().PARTID_MAX),
                                                    Unpredictable_OUT_OF_RANGE_ID);
        assert c IN {Constraint_DEFAULT_ID, Constraint_ANY_ID};
        case c of
            when Constraint_DEFAULT_ID => partidel = DEFAULT_PARTID;
            when Constraint_ANY_ID     => partidel = partid[15:0];
        end;
    end;
    if MPAMIsVirtual(el, mpamdata) then
        return MAP_vPARTID(partidel, partid_err);
    else
        return (partidel, partid_err);
    end;
end;
```
