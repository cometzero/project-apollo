##### J1.4.802 ContiguousSize

```
// ContiguousSize()
// ================
// Return the number of entries log 2 marking a contiguous output range
func ContiguousSize(d128 : bit, tgx : TGx, walkstate : TTWState) => integer
begin
    if d128 == ‘1’ then
        case tgx of
            when TGx_4KB =>
                assert walkstate.level IN {1, 2, 3};
                return if walkstate.level == 1 then 2 else 4;
            when TGx_16KB =>
                assert walkstate.level IN {1, 2, 3};
                if walkstate.level == 1 then
                    return 2;
                elsif walkstate.level == 2 then
                    return 4;
                else
                    return 6;
                end;
            when TGx_64KB =>
                assert walkstate.level IN {2, 3};
                return if walkstate.level == 2 then 6 else 4;
        end;
    end;
    case tgx of
        when TGx_4KB =>
            assert walkstate.level IN {1, 2, 3};
            return 4;
        when TGx_16KB =>
            assert walkstate.level IN {2, 3};
            return if walkstate.level == 2 then 5 else 7;
        when TGx_64KB =>
            assert walkstate.level IN {2, 3};
            return 5;
    end;
end;
```
