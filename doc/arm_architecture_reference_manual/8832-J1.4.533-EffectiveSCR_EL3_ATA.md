##### J1.4.533 EffectiveSCR\_EL3\_ATA

```
// EffectiveSCR_EL3_ATA()
// ======================
// Returns the Effective value of SCR_EL3().ATA.
readonly func EffectiveSCR_EL3_ATA() => bit
begin
    assert !ELUsingAArch32(EL3);
    assert IsFeatureImplemented(FEAT_MTE2);
    if !HaveEL(EL3) then return ‘1’; end;
    return SCR_EL3().ATA;
end;
```
