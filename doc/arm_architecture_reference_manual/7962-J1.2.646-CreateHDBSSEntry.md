##### J1.2.646 CreateHDBSSEntry

```
// CreateHDBSSEntry()
// ==================
// Returns a HDBSS entry.
func CreateHDBSSEntry(ipa : FullAddress, ss : SecurityState, level : integer) => bits(64)
begin
    let ns_ipa : bit = if ss == SS_Secure && ipa.paspace == PAS_NonSecure then ‘1’ else ‘0’;
    return ZeroExtend{64}(ipa.address[NUM_PABITS-1:12] ::
                          ns_ipa :: Zeros{7} :: level[2:0] :: ‘1’);
end;
```
