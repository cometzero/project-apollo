##### J1.4.574 EffectiveSCR\_EL3\_TLOR

```
// EffectiveSCR_EL3_TLOR()
// =======================
// Returns the Effective value of SCR_EL3().TLOR.
readonly func EffectiveSCR_EL3_TLOR() => bit
begin
    assert !ELUsingAArch32(EL3);
    assert IsFeatureImplemented(FEAT_LOR);
    if !HaveEL(EL3) then return ‘0’; end;

end;
```
