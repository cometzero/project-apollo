##### J1.2.609 AArch64\_EffectiveBroadcast

```
// AArch64_EffectiveBroadcast()
// ============================
func AArch64_EffectiveBroadcast(broadcast_in : Broadcast) => Broadcast
begin
    var broadcast : Broadcast = broadcast_in;
    if PSTATE.EL != EL1 then return broadcast; end;
    if broadcast == Broadcast_NSH && EL2Enabled() && HCR_EL2().FB == ‘1’ then
        broadcast = Broadcast_ForcedISH;
    end;
    return broadcast;
end;
```
