##### J1.2.586 BTypeCompatible\_PAC

```
// BTypeCompatible_PAC()
// =====================
// Returns TRUE if PACIxSP or PACIxSPPC instructions are implicitly compatible with PSTATE.BTYPE,
// FALSE otherwise.
func BTypeCompatible_PAC(pacinst : PACInstType) => boolean
begin
    if PSTATE.BTYPE != ‘11’ then
        return TRUE;
    else
        let index : integer = if PSTATE.EL == EL0 then 35 else 36;
        return SCTLR_ELx()[index] == ‘0’;
    end;
end;
```
