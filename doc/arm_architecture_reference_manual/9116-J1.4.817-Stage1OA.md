##### J1.4.817 Stage1OA

```
// Stage1OA()
// ==========
// Given the final walk state (a page or block descriptor), map the untranslated
// input address bits to the output address
func Stage1OA(ia : bits(64), walkparams : S1TTWParams, walkstate : TTWState) => FullAddress
begin
    // Output Address
    var oa : FullAddress;
    let tsize : integer = TranslationSize(walkparams.d128, walkparams.tgx, walkstate.level);
    var csize : integer;
    if walkstate.contiguous == ‘1’ then
        csize = ContiguousSize(walkparams.d128, walkparams.tgx, walkstate);
    else
        csize = 0;
    end;
    let ia_msb : AddressSize = (tsize + csize) as AddressSize;
    oa.paspace = walkstate.baseaddress.paspace;
    oa.address = walkstate.baseaddress.address[NUM_PABITS-1:ia_msb]::ia[ia_msb-1:0];
    return oa;
end;
```
