##### J1.4.767 AbovePPS

```
// AbovePPS()
// ==========
// Returns TRUE if an address exceeds the range configured in GPCCR_EL3.PPS.
func AbovePPS(address : bits(NUM_PABITS)) => boolean
begin
    let pps : integer{} = DecodePPS();
    if pps >= NUM_PABITS then
        return FALSE;
    end;
    return !IsZero(address[NUM_PABITS-1:pps]);
end;
```
