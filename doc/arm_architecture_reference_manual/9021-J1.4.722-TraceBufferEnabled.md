##### J1.4.722 TraceBufferEnabled

```
// TraceBufferEnabled()
// ====================
func TraceBufferEnabled() => boolean
begin
    if !IsFeatureImplemented(FEAT_TRBE) then
        return FALSE;
    elsif SelfHostedTraceEnabled() then
        if TRBLIMITR_EL1().E == ‘0’ then
            return FALSE;
        end;
        var el : bits(2);
        (-, el) = TraceBufferOwner();
        return !ELUsingAArch32(el);
    elsif IsFeatureImplemented(FEAT_TRBE_EXT) then
        return TRBLIMITR_EL1().XE == ‘1’;
    else
        return FALSE;
    end;
end;
```
