// CheckValidStateMatch()
// ======================
// Checks for an invalid state match that will generate Constrained
// Unpredictable behavior, otherwise returns Constraint_NONE.
func CheckValidStateMatch(ssc : bits(2), ssce : bit,
                          hmc : bit, pxc : bits(2),
                          isbreakpnt : boolean) => (Constraint, bits(2), bit, bit, bits(2))
begin
    if !IsFeatureImplemented(FEAT_RME) then assert ssce == ‘0’; end;
    var reserved : boolean = FALSE;
    // Values that are not allocated in any architecture version
    case hmc::ssce::ssc::pxc of
        when ‘0 0 11 10’ => reserved = TRUE;
        when ‘0 1 01 00’ => reserved = TRUE;
        when ‘1 0 00 x0’ => reserved = TRUE;
        when ‘1 0 01 10’ => reserved = TRUE;
        when ‘1 0 1x 10’ => reserved = TRUE;
        when ‘1 1 01 10’ => reserved = TRUE;
        when ‘x 1 x0 xx’ => reserved = TRUE;
        when ‘x 1 11 xx’ => reserved = TRUE;
        otherwise => reserved = FALSE;
    end;
    // Reserved when any combination has HMC = 0, SSCE = 0, and SSC = 0b10.
    if !IsFeatureImplemented(FEAT_Secure) && hmc::ssce::ssc == ‘0 0 10’ then
        reserved = TRUE;
    end;
    // Reserved when any combination has HMC = 0 and SSC = 0b01,
    // and when HMC = 1, SSC = 0b10, and PxC = 0b00.
    if (!IsFeatureImplemented(FEAT_RME) && !IsFeatureImplemented(FEAT_Secure) &&
          (hmc::ssc == ‘0 01’ || hmc::ssc::pxc == ‘1 10 00’)) then
        reserved = TRUE;
    end;
    // Reserved when SSC is either 0b01 or 0b10, when EL3 is not implemented and
    // EL2 is implemented.
    if !HaveEL(EL3) && HaveEL(EL2) && ssc IN {‘01’,‘10’} then
        reserved = TRUE;
    end;
    // Reserved when either HMC = 1 or SSC is not 0b00, when both EL3 and
    // EL2 are not implemented.
    if !HaveEL(EL3) && !HaveEL(EL2) && (hmc == ‘1’ || ssc != ‘00’) then
        reserved = TRUE;
    end;
    // Reserved when HMC = 1, SSC = 0b11, and PxC = 0b00, when EL2 is not implemented.
    if !HaveEL(EL2) && hmc::ssc::pxc == ‘1 11 00’ then
        reserved = TRUE;
    end;
    // Reserved when SSC = 0b11, except when HMC = 1, SSCE = 0, SSC = 0b11, and PxC = 0b00, when
    // Secure EL2 is not implemented.
    if (!IsFeatureImplemented(FEAT_SEL2) && ssc == ‘11’ &&
          hmc::ssce::ssc::pxc != ‘1 0 11 00’) then
        reserved = TRUE;
    end;
    // Reserved when HMC = 1, SSCE = 0, SSC = 0b01 and PxC = 0b00, when EL3 is not implemented OR
    // Secure EL2 is not implemented.
    if ((!HaveEL(EL3) || !IsFeatureImplemented(FEAT_SEL2)) &&
          hmc::ssce::ssc::pxc == ‘1 0 01 00’) then
        reserved = TRUE;
    end;
    // Reserved when HMC = 1, SSC = 0b10, and PxC = 0b00, when AArch64 is not implemented.
    if !HaveAArch64() && hmc::ssc::pxc == ‘1 10 00’ then
        reserved = TRUE;
    end;
    // ‘Usr/Sys/Svc’ encoding is valid only for AArch32 EL1 breakpoints.
    // Otherwise reserved.
    if (!HaveAArch32EL(EL1) || !isbreakpnt) && hmc::pxc == ‘000’ && ssc != ‘11’ then
        reserved = TRUE;
    end;
    if reserved then
        // If parameters are set to a reserved type, behaves as either disabled or a defined type
        var c : Constraint;
        var unpred_state_bits : bits(6);
        var unpred_hmc  : bit;
        var unpred_ssc  : bits(2);
        var unpred_ssce : bit;
        var unpred_pxc  : bits(2);
        (c, unpred_state_bits) = ConstrainUnpredictableBits{6}(Unpredictable_RESBPWPCTRL);
        assert c IN {Constraint_DISABLED, Constraint_UNKNOWN};
        if c == Constraint_DISABLED then
            unpred_hmc  = ARBITRARY : bit;
            unpred_ssc  = ARBITRARY : bits(2);
            unpred_ssce = ARBITRARY : bit;
            unpred_pxc  = ARBITRARY : bits(2);
            return (c, unpred_ssc, unpred_ssce, unpred_hmc, unpred_pxc);
        else
            unpred_hmc  = unpred_state_bits[5];
            unpred_ssc  = unpred_state_bits[4:3];
            unpred_ssce = unpred_state_bits[2];
            unpred_pxc  = unpred_state_bits[1:0];
            return (Constraint_NONE, unpred_ssc, unpred_ssce, unpred_hmc, unpred_pxc);
        end;
    end;
    return (Constraint_NONE, ssc, ssce, hmc, pxc);
end;
