##### J1.4.391 CreateAccDescDCZero

```
// CreateAccDescDCZero()
// =====================
// Access descriptor for data cache zero operations
func CreateAccDescDCZero(cachetype : CacheType) => AccessDescriptor
begin
    var accdesc : AccessDescriptor = NewAccDesc(AccessType_DCZero);
    accdesc.write            = TRUE;
    accdesc.pan              = TRUE;
    accdesc.tagchecked       = cachetype == CacheType_Data;
    accdesc.tagaccess        = cachetype IN {CacheType_Tag,
                                             CacheType_Data_Tag};
    accdesc.cachetype        = cachetype;
    return accdesc;
end;
```
