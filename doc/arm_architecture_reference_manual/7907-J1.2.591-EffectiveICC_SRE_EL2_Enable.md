##### J1.2.591 EffectiveICC\_SRE\_EL2\_Enable

```
// EffectiveICC_SRE_EL2_Enable()
// =============================
// Returns the Effective value of ICC_SRE_EL2.Enable
func EffectiveICC_SRE_EL2_Enable() => bit
begin
    if (ImpDefBool(“Treat ICC_SRE_EL2.SRE as RAO/WI”) &&
          ImpDefBool(“Treat ICC_SRE_EL2.Enable as RAO/WI”)) then
        return ‘1’;
    end;
    if ICC_SRE_EL2().SRE == ‘0’ then
        return ‘1’;
    end;
    return ICC_SRE_EL2().Enable;
end;
```
