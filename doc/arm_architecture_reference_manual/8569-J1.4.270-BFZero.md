##### J1.4.270 BFZero

```
// BFZero()
// ========
func BFZero{N}(sign : bit) => bits(N)
begin
    assert N == 16;
    let E : integer{} = 8;
    let F : integer{} = N - (E + 1);
    return sign :: Zeros{E} :: Zeros{F};
end;
```
