// ELR_EL - accessor
// =================
accessor ELR_EL(el : bits(2)) <=> value : bits(64)
begin
    getter

        case el of
            when EL1 =>  r = ELR_EL1();

    end;
    setter
        let r : bits(64) = value;
        case el of
            when EL1 =>  ELR_EL1() = r;
            when EL2 =>  ELR_EL2() = r;
            when EL3 =>  ELR_EL3() = r;
            otherwise => unreachable;
        end;
    end;
end;
