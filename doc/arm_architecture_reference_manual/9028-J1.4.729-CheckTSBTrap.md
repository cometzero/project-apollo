##### J1.4.729 CheckTSBTrap

```
// CheckTSBTrap()
// ==============
// Checks whether TSB should trap to EL2.
func CheckTSBTrap()
begin
    if (IsFeatureImplemented(FEAT_FGT2) && IsFeatureImplemented(FEAT_TRBEv1p1) &&
          PSTATE.EL IN {EL0, EL1} && EL2Enabled() && !IsInHost() &&
          (!HaveEL(EL3) || SCR_EL3().FGTEn2 == ‘1’) && HFGITR2_EL2().TSBCSYNC == ‘1’) then
        let target_el : bits(2) = EL2;
        let iss : bits(25)      = 0x4[24:0];
        AArch64_OtherInstrTrap(target_el, iss);
    end;
end;
func TraceSynchronizationBarrier()
begin
    if IsFeatureImplemented(FEAT_TRBE) && !TraceAllowed(PSTATE.EL) then
        TraceUnitFlush();
    end;
    return;
end;
```
