##### J1.4.786 GPTGranulesDescriptorValid

```
// GPTGranulesDescriptorValid()
// ============================
// Returns TRUE if the given GPT Granules descriptor is valid, and FALSE otherwise.
func GPTGranulesDescriptorValid(level_1_entry : bits(64)) => boolean
begin
    for i = 0 to 15 do
        if !GPIValid(level_1_entry[i*4 +:4]) then
            return FALSE;
        end;
    end;
    return TRUE;
end;
```
