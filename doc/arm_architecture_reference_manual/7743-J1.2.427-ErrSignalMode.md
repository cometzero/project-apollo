```
// ErrSignalMode
// =============
// Implementation options for the ANERR/ADERR synchronous error behavior.

    ErrSignalMode_NotImplemented,
    ErrSignalMode_EnSelfOnly,        // EnANERR-only / EnADERR-only

};
```
