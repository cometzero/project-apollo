##### J1.2.561 SCTLR\_EL

```
// SCTLR_EL - accessor
// ===================
accessor SCTLR_EL(regime : bits(2)) <=> value : SCTLRType
begin
    getter
        var r : bits(64);
        case regime of
            when EL1 =>  r = SCTLR_EL1();
            when EL2 =>  r = SCTLR_EL2();
            when EL3 =>  r = SCTLR_EL3();
            otherwise => unreachable;
        end;
        return r;
    end;
    setter
        let r : bits(64) = value;
        case regime of
            when EL1 => SCTLR_EL1() = r;
            when EL2 => SCTLR_EL2() = r;
            when EL3 => SCTLR_EL3() = r;
            otherwise => unreachable;
        end;
    end;
end;
```
