##### J1.4.418 CreateAccDescTTEUpdate

// CreateAccDescTTEUpdate()
// ========================
// Access descriptor for translation table entry HW update
func CreateAccDescTTEUpdate(accdesc_in : AccessDescriptor) => AccessDescriptor
begin
    var accdesc : AccessDescriptor = NewAccDesc(AccessType_TTW, accdesc_in.mpam);
    accdesc.el              = accdesc_in.el;
    accdesc.ss              = accdesc_in.ss;
    accdesc.atomicop        = TRUE;
    accdesc.modop           = MemAtomicOp_CAS;
    accdesc.read            = TRUE;
    accdesc.write           = TRUE;
    return accdesc;
end;
