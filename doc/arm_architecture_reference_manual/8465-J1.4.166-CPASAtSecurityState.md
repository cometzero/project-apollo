##### J1.4.166 CPASAtSecurityState

```
// CPASAtSecurityState()
// =====================
// Get cache PA space for given security state.
func CPASAtSecurityState(ss : SecurityState) => CachePASpace
begin
    case ss of
        when SS_NonSecure =>
            return CPAS_NonSecure;
        when SS_Secure =>
            return CPAS_SecureNonSecure;
        when SS_Root =>
            return CPAS_Any;
        when SS_Realm =>
            return CPAS_RealmNonSecure;
    end;
end;
```
