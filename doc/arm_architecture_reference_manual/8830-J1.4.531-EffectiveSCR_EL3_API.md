##### J1.4.531 EffectiveSCR\_EL3\_API

// EffectiveSCR_EL3_API()
// ======================
// Returns the Effective value of SCR_EL3().API.
readonly func EffectiveSCR_EL3_API() => bit
begin
    assert !ELUsingAArch32(EL3);
    assert IsFeatureImplemented(FEAT_PAuth);
    if !HaveEL(EL3) then return ‘1’; end;

end;
