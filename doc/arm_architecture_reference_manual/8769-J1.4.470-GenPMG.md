# ​J1.4.470 GenPMG

##### J1.4.470 GenPMG

```
// GenPMG()
// ========
// Returns PMG for Exception level el.
// if mpamdata.sm is TRUE then PMG is from MPAMSM_EL1.
// Otherwise, PMG is from MPAMn_ELx.
// If PARTID generation (GenPARTID) encountered an error, GenPMG() should be
// called with partid_err as TRUE.
func GenPMG(el : bits(2), mpamdata : MPAMdata, partid_err : boolean) => PMGType
begin
    if partid_err && ConstrainUnpredictableBool(Unpredictable_USE_DEFAULT_PMG) then
        return DEFAULT_PMG;
    end;
    let groupel : PMGType = GetMPAM_PMG(el, mpamdata);
    if UInt(groupel) <= UInt(MPAMIDR_EL1().PMG_MAX) then
        return groupel;
    end;
    return DEFAULT_PMG;
end;
```
