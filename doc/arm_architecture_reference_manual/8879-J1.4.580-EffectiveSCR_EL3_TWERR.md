##### J1.4.580 EffectiveSCR\_EL3\_TWERR

```
// EffectiveSCR_EL3_TWERR()
// ========================
// Returns the Effective value of SCR_EL3().TWERR.
readonly func EffectiveSCR_EL3_TWERR() => bit
begin
    assert !ELUsingAArch32(EL3);
    assert IsFeatureImplemented(FEAT_RASv2);
    if !HaveEL(EL3) then return ‘0’; end;
    return SCR_EL3().TWERR;
end;
```
